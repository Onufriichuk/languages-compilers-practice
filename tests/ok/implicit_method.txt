struct Point
{
 i32 mut x
 i32 mut y
 fn d2 := () -> i32
 {
  i32 r{x * x + y * y}
  exit r
 }
 fn shift := (i32 dx) -> i32
 {
  x := x + dx
  exit x
 }
 fn add := (i32 dx, i32 dy) -> i32
 {
  x := x + dx
  y := y + dy
  exit x + y
 }
 fn again := () -> i32
 {
  i32 a{d2()}
  exit a
 }
}
Point p{3, 4}
i32 a{p.again()}
exit a
